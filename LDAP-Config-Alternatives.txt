# LDAP Konfigürasyon Alternatifleri
# Aşağıdaki formatlardan birini test edin

## ============================================
## SENARYO 1: Domain = boerltd.local (Standart)
## ============================================
{
  "Authentication": {
	"Ldap": {
	  "Host": "192.168.0.17",
	  "Port": 389,
	  "Domain": "boerltd.local",
	  "BaseDn": "DC=boerltd,DC=local",
	  "SearchFilter": "(sAMAccountName={0})",
	  "UseSsl": false
	}
  }
}

# Giriş formatları:
# - administrator
# - boerltd\administrator
# - administrator@boerltd.local

## TEST KOMUTLARI
## ============================================

# PowerShell ile domain bilgisi almak:
Get-ADDomain -Server 192.168.0.17 | Select-Object DNSRoot, NetBIOSName, DistinguishedName

# Ping testi:
Test-NetConnection -ComputerName 192.168.0.17 -Port 389

# LDAP bağlantı testi (PowerShell):
$ldap = New-Object System.DirectoryServices.Protocols.LdapConnection("192.168.0.17:389")
$ldap.AuthType = [System.DirectoryServices.Protocols.AuthType]::Basic
$credential = New-Object System.Net.NetworkCredential("boerltd\administrator", "YourPassword")
$ldap.Credential = $credential
$ldap.Bind()
